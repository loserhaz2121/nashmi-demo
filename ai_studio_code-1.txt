class FareEstimate {
  final double distanceKm;
  final int durationMin;
  final double motoFareJod;
  final double carAppsMinFareJod;
  final double savingsPercent;

  FareEstimate({
    required this.distanceKm,
    required this.durationMin,
    required this.motoFareJod,
    required this.carAppsMinFareJod,
    required this.savingsPercent,
  });
}

class JordanPricingService {
  static const double motoBaseFare = 0.25;
  static const double motoPerKm = 0.16;
  static const double motoPerMin = 0.03;
  static const double motoMinFare = 0.85;

  static const double carBaseDay = 0.45;
  static const double carPerKmDay = 0.316;
  static const double carPerMinDay = 0.06;

  static FareEstimate calculate({required double distanceKm, required int durationMin}) {
    double rawMoto = motoBaseFare + (distanceKm * motoPerKm) + (durationMin * motoPerMin);
    double finalMoto = rawMoto < motoMinFare ? motoMinFare : rawMoto;
    finalMoto = (finalMoto * 20).round() / 20.0;

    double carFare = carBaseDay + (distanceKm * carPerKmDay) + (durationMin * carPerMinDay);
    if (carFare < 1.50) carFare = 1.50;
    double savings = ((carFare - finalMoto) / carFare) * 100;

    return FareEstimate(
      distanceKm: double.parse(distanceKm.toStringAsFixed(1)),
      durationMin: durationMin,
      motoFareJod: double.parse(finalMoto.toStringAsFixed(2)),
      carAppsMinFareJod: double.parse(carFare.toStringAsFixed(2)),
      savingsPercent: savings > 0 ? savings.roundToDouble() : 0,
    );
  }
}

class RouteResult {
  final List<LatLng> points;
  final double distanceKm;
  final int durationMin;
  RouteResult({required this.points, required this.distanceKm, required this.durationMin});
}

class RoutingService {
  static Future<RouteResult> getRoute(LatLng start, LatLng end) async {
    final url = Uri.parse(
      'https://router.project-osrm.org/route/v1/driving/'
      '${start.longitude},${start.latitude};${end.longitude},${end.latitude}'
      '?overview=full&geometries=geojson',
    );
    try {
      final response = await http.get(url).timeout(const Duration(seconds: 6));
      if (response.statusCode == 200) {
        final data = jsonDecode(response.body);
        final route = data['routes'][0];
        final double distKm = (route['distance'] as num).toDouble() / 1000.0;
        final int durMin = ((route['duration'] as num).toDouble() / 60.0 * 0.85).ceil();
        final coords = route['geometry']['coordinates'] as List;
        final points = coords.map((c) => LatLng((c[1] as num).toDouble(), (c[0] as num).toDouble())).toList();
        return RouteResult(points: points, distanceKm: distKm, durationMin: durMin < 3 ? 3 : durMin);
      }
    } catch (_) {}
    const distance = Distance();
    final directKm = distance.as(LengthUnit.Kilometer, start, end) * 1.25;
    return RouteResult(points: [start, end], distanceKm: directKm < 0.5 ? 0.5 : directKm, durationMin: (directKm * 2.5).ceil());
  }
}

class JordanPhoneHelper {
  static String? formatToE164(String input) {
    String cleaned = input.replaceAll(RegExp(r'[\s\-\(\)]'), '');
    if (cleaned.startsWith('+962')) cleaned = cleaned.substring(4);
    else if (cleaned.startsWith('00962')) cleaned = cleaned.substring(5);
    else if (cleaned.startsWith('962')) cleaned = cleaned.substring(3);
    if (cleaned.startsWith('0')) cleaned = cleaned.substring(1);
    if (!RegExp(r'^7[789]\d{7}$').hasMatch(cleaned)) return null;
    return '+962$cleaned';
  }
}

class _TriangleClipper extends CustomClipper<Path> {
  @override
  Path getClip(Size size) => Path()..moveTo(0, 0)..lineTo(size.width, size.height / 2)..lineTo(0, size.height)..close();

class _SevenPointStarPainter extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()..color = JordanColors.white;
    final path = Path();
    final cx = size.width / 2, cy = size.height / 2, outerR = size.width / 2, innerR = outerR * 0.45;
    for (int i = 0; i < 14; i++) {
      final r = i.isEven ? outerR : innerR;
      final angle = (math.pi / 7) * i - math.pi / 2;
      final x = cx + r * math.cos(angle), y = cy + r * math.sin(angle);
      if (i == 0) path.moveTo(x, y); else path.lineTo(x, y);
    }
    path.close();
    canvas.drawPath(path, paint);
  }
  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
}

class _CaptainHomeScreenState extends State<CaptainHomeScreen> {
  final String _captainId = FirebaseAuth.instance.currentUser!.uid;
  final MapController _mapController = MapController();

  bool _isOnline = false;
  LatLng _myLocation = const LatLng(31.9600, 35.9150);
  String? _activeRideId;
  List<LatLng> _activeRoute = [];
  double _todayEarnings = 0.0;
  StreamSubscription<Position>? _posSub;

  @override
  void dispose() { _posSub?.cancel(); super.dispose(); }

  Future<void> _toggleOnline(bool val) async {
    if (val) {
      final ok = await showDialog<bool>(
        context: context,
        builder: (_) => AlertDialog(
          title: const Text('تأكيد السلامة'),
          content: const Text('هل ترتدي الخوذة وتتوفر معك خوذة للراكب؟'),
          actions: [
            TextButton(onPressed: () => Navigator.pop(context, false), child: const Text('لا')),
            ElevatedButton(onPressed: () => Navigator.pop(context, true), child: const Text('نعم')),
          ],
        ),
      );
      if (ok != true) return;
      await _startLocation();
      await FirebaseFirestore.instance.collection('captains_online').doc(_captainId).set({
        'captainId': _captainId, 'name': 'كابتن نشمي', 'lat': _myLocation.latitude, 'lng': _myLocation.longitude, 'isBusy': false,
      });
    } else {
      _posSub?.cancel();
      await FirebaseFirestore.instance.collection('captains_online').doc(_captainId).delete();
    }
    setState(() => _isOnline = val);
  }

  Future<void> _startLocation() async {
    try {
      LocationPermission p = await Geolocator.checkPermission();
      if (p == LocationPermission.denied) p = await Geolocator.requestPermission();
      if (p == LocationPermission.always || p == LocationPermission.whileInUse) {
        final pos = await Geolocator.getCurrentPosition();
        setState(() => _myLocation = LatLng(pos.latitude, pos.longitude));
        _mapController.move(_myLocation, 15);
        _posSub = Geolocator.getPositionStream(locationSettings: const LocationSettings(distanceFilter: 15)).listen((pt) {
          setState(() => _myLocation = LatLng(pt.latitude, pt.longitude));
          if (_isOnline) FirebaseFirestore.instance.collection('captains_online').doc(_captainId).update({'lat': pt.latitude, 'lng': pt.longitude});
        });
      }
    } catch (_) {}
  }

  Future<void> _acceptRide(Map<String, dynamic> ride) async {
    final id = ride['rideId'] as String;
    await FirebaseFirestore.instance.collection('rides').doc(id).update({'status': 'accepted', 'captainId': _captainId, 'captainName': 'كابتن نشمي'});
    await FirebaseFirestore.instance.collection('captains_online').doc(_captainId).update({'isBusy': true});
    final route = await RoutingService.getRoute(_myLocation, LatLng(ride['pickupLat'], ride['pickupLng']));
    setState(() { _activeRideId = id; _activeRoute = route.points; });
  }

  Future<void> _updateStatus(String next, Map<String, dynamic> ride) async {
    await FirebaseFirestore.instance.collection('rides').doc(_activeRideId).update({'status': next});
    if (next == 'started') {
      final r = await RoutingService.getRoute(LatLng(ride['pickupLat'], ride['pickupLng']), LatLng(ride['dropoffLat'], ride['dropoffLng']));
      setState(() => _activeRoute = r.points);
    } else if (next == 'completed') {
      final fare = (ride['fareJod'] as num).toDouble();
      await FirebaseFirestore.instance.collection('captains_online').doc(_captainId).update({'isBusy': false});
      setState(() { _todayEarnings += fare * 0.90; _activeRideId = null; _activeRoute = []; });
    }
  }

class AuthGate extends StatelessWidget {
  const AuthGate({super.key});
  @override
  Widget build(BuildContext context) {
    return StreamBuilder<User?>(
      stream: FirebaseAuth.instance.authStateChanges(),
      builder: (context, authSnap) {
        if (authSnap.connectionState == ConnectionState.waiting) return const Scaffold(body: Center(child: CircularProgressIndicator()));
        final user = authSnap.data;
        if (user == null) return const PhoneLoginScreen();
        return FutureBuilder<DocumentSnapshot>(
          future: FirebaseFirestore.instance.collection('users').doc(user.uid).get(),
          builder: (context, userSnap) {
            if (!userSnap.hasData) return const Scaffold(body: Center(child: CircularProgressIndicator()));
            final data = userSnap.data!.data() as Map<String, dynamic>?;
            if (data == null) return const PhoneLoginScreen();
            return data['role'] == 'captain' ? const CaptainHomeScreen() : const ClientHomeScreen();
          },
        );
      },
    );
  }
